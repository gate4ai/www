GATE4AI_VERSION=v0.1.1
{{if eq .Page.Lang.Code "ru"}}конкретная версия вместо последней{{else}}a specific version instead of the latest{{end}}
GATE4AI_INSTALL_DIR=…
{{if eq .Page.Lang.Code "ru"}}каталог установки (по умолчанию ~/.local/bin, на Windows %LOCALAPPDATA%\Programs\gate4ai){{else}}install folder (default ~/.local/bin, on Windows %LOCALAPPDATA%\Programs\gate4ai){{end}}
GATE4AI_NO_TRAY=1
{{if eq .Page.Lang.Code "ru"}}без значка в трее (серверы без графики){{else}}skip the tray icon app (headless servers){{end}}
GATE4AI_NO_START=1
{{if eq .Page.Lang.Code "ru"}}не включать автозапуск (gate4ai service install); запущенный daemon при обновлении всё равно перезапускается{{else}}do not turn on the autostart (gate4ai service install); a running daemon is still restarted on update{{end}}
GATE4AI_NO_OPEN=1
{{if eq .Page.Lang.Code "ru"}}не открывать мастер в браузере после установки (и так не открывается без графической сессии и при заданной CI){{else}}do not open the setup wizard in the browser after installing (it is not opened anyway without a graphical session or with CI set){{end}}
GATE4AI_INSTALL_DEPS=1
{{if eq .Page.Lang.Code "ru"}}поставить для браузера агентов Google Chrome, а на Linux без графики ещё виртуальный экран Xvfb и шрифты — без вопроса (по умолчанию gate4ai browser setup проверяет и спрашивает в терминале){{else}}install Google Chrome for the browser of agents, and on Linux without a screen also the Xvfb virtual screen and fonts, without asking (by default gate4ai browser setup checks them and asks at the terminal){{end}}
GATE4AI_NO_BROWSER=1
{{if eq .Page.Lang.Code "ru"}}без браузера агентов: не проверять Chrome и не ставить пакет browser{{else}}no browser for agents: do not check Chrome and do not install the browser package{{end}}
curl -fsSL https://gate4.ai/install.sh | GATE4AI_NO_TRAY=1 GATE4AI_NO_OPEN=1 GATE4AI_INSTALL_DEPS=1 sh
{{end}}
